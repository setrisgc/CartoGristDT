// ====================================== // RECEPTION DES DONNEES GRIST // ====================================== grist.onRecords(function(records){ couchePVD.clearLayers(); coucheACV.clearLayers(); coucheVA.clearLayers(); couchecommunesinteressees.clearLayers(); coucheAutres.clearLayers(); let bounds=[]; records.forEach(row=>{ if( row.latitude && row.longitude ){ let lat=parseFloat(row.latitude); let lon=parseFloat(row.longitude); if( !isNaN(lat) && !isNaN(lon) ){ // ================================ // COMMUNE INTERESSEE // ================================ if( row.communes_interessees && row.communes_interessees.toString().trim().toLowerCase() === "oui" ){ let interesse = L.circleMarker( [lat,lon], { pane:'interesseesPane', radius:8, color:"#ff0000", weight:3, fillColor:"#ff0000", fillOpacity:0.15 } ); interesse.bindPopup(contenuPopup(row)); interesse.addTo(couchecommunesinteressees); } // ================================ // COMMUNE CONTRACTUALISEE // ================================ let contrat = (row.contractualisation || "") .trim() .toUpperCase(); if( contrat==="PVD" || contrat==="ACV" || contrat==="VA" ){ let point = L.circleMarker( [lat,lon], { pane:'cmcommPane', radius:7, color:"transparent", weight:0, fillColor: couleurcontractualisation(row.contractualisation), fillOpacity:0.9 } ); point.bindPopup(contenuPopup(row)); switch(contrat){ case "PVD": point.addTo(couchePVD); break; case "ACV": point.addTo(coucheACV); break; case "VA": point.addTo(coucheVA); break; } } else { let point = L.circleMarker( [lat, lon], { pane: 'cmcommPane', radius: 6, color: "#000000", weight: 2, fillColor: "#ffffff", fillOpacity: 0.1 } ); point.bindPopup(contenuPopup(row)); point.addTo(coucheAutres); } // ================================ // AJOUT POUR LE ZOOM // ================================ bounds.push([lat,lon]); } // fermeture if !isNaN } // fermeture if latitude longitude }); // fermeture records.forEach // zoom automatique if(bounds.length>0 && !map._initialZoom){ map.fitBounds(bounds); map._initialZoom=true; } }); // <-- fermeture grist.onRecords
